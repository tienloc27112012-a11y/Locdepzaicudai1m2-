-- 3.1 Tắt hiệu ứng không cần thiết
local function OptimizeGraphics()
    -- Giảm chất lượng đồ họa
    if Lighting then
        Lighting.GlobalShadows = false
        Lighting.FogEnd = 1000
        Lighting.FogStart = 500
        Lighting.Brightness = 1
        Lighting.ClockTime = 12
        Lighting.Ambient = Color3.fromRGB(100, 100, 100)
        Lighting.OutdoorAmbient = Color3.fromRGB(100, 100, 100)
    end
    
    -- Tắt hiệu ứng hạt
    for _, v in pairs(Workspace:GetDescendants()) do
        if v:IsA("ParticleEmitter") or v:IsA("Trail") or v:IsA("Fire") or v:IsA("Smoke") or v:IsA("Sparkles") then
            v.Enabled = false
            v:Destroy()
        end
        if v:IsA("Decal") or v:IsA("Texture") then
            v.Transparency = 1
        end
        if v:IsA("BasePart") and not v:IsA("MeshPart") then
            v.Material = Enum.Material.Plastic
            v.Reflectance = 0
        end
    end
    
    -- Giảm khoảng cách hiển thị
    Workspace.CurrentCamera.CameraSubject = Character.Humanoid
    Workspace.CurrentCamera.FieldOfView = 50
end

-- 3.2 Xóa vật thể rác khỏi bộ nhớ
local function CleanupMemory()
    local toRemove = {}
    for _, v in pairs(Workspace:GetDescendants()) do
        if v:IsA("Model") and not v:FindFirstChild("Humanoid") and not v:IsA("Terrain") and not v:IsA("Folder") then
            if v:FindFirstChild("Handle") and v:FindFirstChild("TouchInterest") then
                -- Giữ lại vật phẩm có thể nhặt
            else
                table.insert(toRemove, v)
            end
        end
        -- Xóa các âm thanh dư thừa
        if v:IsA("Sound") and v.IsPlaying then
            v:Stop()
            v:Destroy()
        end
        -- Xóa các GUI rác
        if v:IsA("ScreenGui") and v ~= CoreGui:FindFirstChild("PermanentText") then
            if not v:FindFirstChild("MainFrame") and not v.Name:find("Roblox") then
                table.insert(toRemove, v)
            end
        end
    end
    for _, v in pairs(toRemove) do
        v:Destroy()
    end
    task.wait(0.5)
    collectgarbage("collect")
end

-- 3.3 Tối ưu vòng lặp render
local function OptimizeRenderLoop()
    -- Giảm tần suất render khi không cần thiết
    RunService:SetThrottleFramerate(30) -- Giới hạn 30 FPS để giảm tải CPU
    RunService.RenderStepped:Connect(function()
        -- Giảm tần suất cập nhật animation nếu xa
        for _, v in pairs(Workspace:GetDescendants()) do
            if v:IsA("AnimationTrack") and v.IsPlaying then
                if v.Looped and v.Length > 5 then
                    v:Stop()
                end
            end
        end
    end)
end

-- 3.4 Tắt chế độ nhạy cảm va chạm
local function OptimizePhysics()
    Workspace.Gravity = 100
    Workspace.FallenPartsDestroyHeight = -1000
    Workspace.Stepped:Connect(function()
        for _, v in pairs(Workspace:GetDescendants()) do
            if v:IsA("BasePart") and v.Anchored == false and v.Velocity.Magnitude > 500 then
                v.Anchored = true
                Debris:AddItem(v, 2)
            end
        end
    end)
end

-- 3.5 Xóa tin nhắn chat và thông báo không cần thiết
local function CleanChatAndNotifications()
    local chat = CoreGui:FindFirstChild("Chat")
    if chat then
        for _, v in pairs(chat:GetDescendants()) do
            if v:IsA("Frame") and v:FindFirstChild("TextLabel") then
                v:Destroy()
            end
        end
    end
    
    -- Xóa thông báo nhiệm vụ
    local notification = CoreGui:FindFirstChild("Notification")
    if notification then
        notification:Destroy()
    end
end

local function GetCurrentLevel()
    local stats = LocalPlayer:FindFirstChild("Data") and LocalPlayer.Data:FindFirstChild("Stats")
    if stats and stats:FindFirstChild("Level") then
        return stats.Level.Value
    end
    return 0
end

local function GetCurrentZone()
    local lv = GetCurrentLevel()
    for _, zone in ipairs(FarmZones) do
        if lv >= zone.min and lv <= zone.max then
            return zone
        end
    end
    return FarmZones[1]
end

local function TeleportTo(pos)
    if not pos then return end
    if (RootPart.Position - pos).Magnitude < 10 then return end
    local tween = TweenService:Create(RootPart, TweenInfo.new(0.6, Enum.EasingStyle.Linear), {CFrame = CFrame.new(pos)})
    tween:Play()
    tween.Completed:Wait()
end

local function MoveTo(pos)
    if not pos then return end
    RootPart.CFrame = CFrame.new(pos)
end

local function GetClosestEnemy(zone)
    local target = nil
    local minDist = 80
    for _, v in pairs(Workspace:GetDescendants()) do
        if v:IsA("Model") and v:FindFirstChild("Humanoid") and v.Humanoid.Health > 0 then
            if v:FindFirstChild("HumanoidRootPart") then
                local isBoss = false
                for _, bossName in ipairs(BossList) do
                    if v.Name:find(bossName) then isBoss = true break end
                end
                local isMob = v.Name:find(zone.mob) or v.Name:find("Boss") or v.Name:find("Guard")
                if isMob or isBoss then
                    local dist = (RootPart.Position - v.HumanoidRootPart.Position).Magnitude
                    if dist < minDist then
                        target = v
                        minDist = dist
                    end
                end
            end
        end
    end
    return target
end

local function GetClosestItem()
    local closest = nil
    local minDist = 120
    for _, v in pairs(Workspace:GetDescendants()) do
        if v:IsA("Model") and v:FindFirstChild("Handle") then
            local dist = (RootPart.Position - v.Handle.Position).Magnitude
            if dist < minDist then
                closest = v
                minDist = dist
            end
        end
    end
    return closest
end

local function AutoQuest()
    for _, npc in pairs(Workspace:GetDescendants()) do
        if npc:IsA("Model") and npc:FindFirstChild("Humanoid") and npc:FindFirstChild("HumanoidRootPart") then
            if npc.Name:find("Quest") or npc.Name:find("Giver") or npc.Name:find("NPC") then
                local dist = (RootPart.Position - npc.HumanoidRootPart.Position).Magnitude
                if dist < 25 then
                    TeleportTo(npc.HumanoidRootPart.Position + Vector3.new(0, 0, 4))
                    task.wait(0.3)
                    VirtualUser:Button1Down(Vector2.new(0, 0))
                    task.wait(0.1)
                    VirtualUser:Button1Up(Vector2.new(0, 0))
                    task.wait(0.5)
                    -- Nhấn E để tương tác
                    VirtualUser:KeyDown(Enum.KeyCode.E)
                    task.wait(0.2)
                    VirtualUser:KeyUp(Enum.KeyCode.E)
                    task.wait(1)
                    -- Click nhận nhiệm vụ
                    VirtualUser:Button1Down(Vector2.new(400, 300))
                    task.wait(0.1)
                    VirtualUser:Button1Up(Vector2.new(400, 300))
                    task.wait(0.3)
                    VirtualUser:Button1Down(Vector2.new(400, 350))
                    task.wait(0.1)
                    VirtualUser:Button1Up(Vector2.new(400, 350))
                    return true
                end
            end
        end
    end
    return false
end

-- [KHỐI 8] Tự động farm quái (có combo và dodge)
-- ============================================================
local function AutoFarming()
    local zone = GetCurrentZone()
    currentArea = zone.name
    
    -- Teleport đến khu vực
    TeleportTo(zone.pos)
    task.wait(0.5)
    
    -- Nhận nhiệm vụ nếu có
    AutoQuest()
    
    while isFarming and task.wait(0.05) do
        if isPaused then task.wait(1); break end
        if isInRaid then task.wait(1); break end
        if isDead then task.wait(3); break end
        
        -- Kiểm tra cập nhật level
        currentLevel = GetCurrentLevel()
        local newZone = GetCurrentZone()
        if newZone.name ~= currentArea then
            currentArea = newZone.name
            TeleportTo(newZone.pos)
            task.wait(0.5)
            AutoQuest()
        end
        
        -- Tìm mục tiêu
        local target = GetClosestEnemy(zone)
        
        if target then
            -- Teleport đến gần mục tiêu
            local targetPos = target.HumanoidRootPart.Position
            MoveTo(targetPos + Vector3.new(0, 0, 3))
            
            -- Combo tấn công (5 đòn liên tiếp)
            for i = 1, 5 do
                VirtualUser:Button1Down(Vector2.new(0, 0))
                task.wait(0.04)
                VirtualUser:Button1Up(Vector2.new(0, 0))
                task.wait(0.04)
                -- Phím E cho skill
                if i % 2 == 0 then
                    VirtualUser:KeyDown(Enum.KeyCode.E)
                    task.wait(0.05)
                    VirtualUser:KeyUp(Enum.KeyCode.E)
                end
                -- Phím Q cho skill 2
                if i % 3 == 0 then
                    VirtualUser:KeyDown(Enum.KeyCode.Q)
                    task.wait(0.05)
                    VirtualUser:KeyUp(Enum.KeyCode.Q)
                end
                -- Phím R cho skill 3
                if i % 4 == 0 then
                    VirtualUser:KeyDown(Enum.KeyCode.R)
                    task.wait(0.05)
                    VirtualUser:KeyUp(Enum.KeyCode.R)
                end
            end
            
            -- Nhặt vật phẩm rơi ra
            local item = GetClosestItem()
            if item then
                MoveTo(item.Handle.Position)
                task.wait(0.2)
                VirtualUser:KeyDown(Enum.KeyCode.E)
                task.wait(0.1)
                VirtualUser:KeyUp(Enum.KeyCode.E)
            end
            
            totalKills = totalKills + 1
            
        else
            -- Không tìm thấy quái, di chuyển xung quanh
            local x = math.random(-30, 30)
            local z = math.random(-30, 30)
            MoveTo(zone.pos + Vector3.new(x, 0, z))
            task.wait(0.5)
            
            -- Nếu vẫn không có quái, thử nhận lại nhiệm vụ
            if totalKills % 20 == 0 then
                AutoQuest()
            end
        end
        
        -- Tự động nhặt vật phẩm xung quanh mỗi 2 giây
        if math.random(1, 20) == 1 then
            local item = GetClosestItem()
            if item then
                MoveTo(item.Handle.Position)
                task.wait(0.2)
                VirtualUser:KeyDown(Enum.KeyCode.E)
                task.wait(0.1)
                VirtualUser:KeyUp(Enum.KeyCode.E)
            end
        end
    end
end

-- [KHỐI 9] Tự động nâng cấp chỉ số với đầy đủ 5 stat

local function AutoUpgradeStats()
    while isFarming and task.wait(15) do
        if isPaused then task.wait(1); break end
        local stats = LocalPlayer:FindFirstChild("Data") and LocalPlayer.Data:FindFirstChild("Stats")
        if stats then
            local priorities = {"Melee", "Defense", "Sword", "Gun", "Fruit"}
            for _, statName in ipairs(priorities) do
                local stat = stats:FindFirstChild(statName)
                if stat and stat:FindFirstChild("Level") and stat:FindFirstChild("MaxLevel") then
                    local pointsAvailable = 0
                    if stats:FindFirstChild("Points") then
                        pointsAvailable = stats.Points.Value
                    end
                    if pointsAvailable > 0 and stat.Level.Value < stat.MaxLevel.Value then
                        local upgradeRequest = ReplicatedStorage:FindFirstChild("UpgradeStat")
                        if upgradeRequest then
                            upgradeRequest:InvokeServer(statName)
                            task.wait(0.3)
                        end
                    end
                end
            end
        end
    end
end

-- ============================================================
local function FindRaidEvent()
    for _, v in pairs(Workspace:GetDescendants()) do
        if v:IsA("Model") and v:FindFirstChild("Humanoid") and v.Humanoid.Health > 0 then
            if v:FindFirstChild("HumanoidRootPart") then
                for _, bossName in ipairs(BossList) do
                    if v.Name:find(bossName) then
                        local dist = (RootPart.Position - v.HumanoidRootPart.Position).Magnitude
                        if dist < 500 then
                            return v
                        end
                    end
                end
                if v.Name:find("Rip") or v.Name:find("Indra") or v.Name:find("Zioles") then
                    return v
                end
            end
        end
    end
    return nil
end

local function JoinRaid()
    while isFarming and task.wait(3) do
        local raidBoss = FindRaidEvent()
        if raidBoss then
            isInRaid = true
            local bossPos = raidBoss.HumanoidRootPart.Position
            TeleportTo(bossPos + Vector3.new(0, 8, 12))
            task.wait(0.3)
            
            -- Đánh boss raid
            while raidBoss and raidBoss.Humanoid.Health > 0 and task.wait(0.08) do
                if not raidBoss.Parent then break end
                if not raidBoss:FindFirstChild("Humanoid") then break end
                local currentPos = raidBoss.HumanoidRootPart.Position
                MoveTo(currentPos + Vector3.new(0, 0, 5))
                
                -- Tấn công liên tục
                for i = 1, 8 do
                    VirtualUser:Button1Down(Vector2.new(0, 0))
                    task.wait(0.03)
                    VirtualUser:Button1Up(Vector2.new(0, 0))
                    task.wait(0.03)
                    VirtualUser:KeyDown(Enum.KeyCode.E)
                    task.wait(0.05)
                    VirtualUser:KeyUp(Enum.KeyCode.E)
                end
                
                -- Nhặt vật phẩm rơi ra từ boss
                for _, item in pairs(Workspace:GetDescendants()) do
                    if item:IsA("Model") and item:FindFirstChild("Handle") then
                        local dist = (RootPart.Position - item.Handle.Position).Magnitude
                        if dist < 40 then
                            MoveTo(item.Handle.Position)
                            task.wait(0.1)
                            VirtualUser:KeyDown(Enum.KeyCode.E)
                            task.wait(0.05)
                            VirtualUser:KeyUp(Enum.KeyCode.E)
                        end
                    end
                end
            end
            
            isInRaid = false
            -- Quay lại farm
            local zone = GetCurrentZone()
            TeleportTo(zone.pos)
            task.wait(0.5)
        end
    end
end

local function DeathHandler()
    while isFarming and task.wait(0.3) do
        if Humanoid.Health <= 0 then
            isDead = true
            task.wait(4) -- Chờ hồi sinh
            
            -- Kiểm tra đã hồi sinh chưa
            local attempts = 0
            while Humanoid.Health <= 0 and attempts < 10 do
                task.wait(1)
                attempts = attempts + 1
            end
            
            isDead = false
            -- Teleport về khu farm
            local zone = GetCurrentZone()
            TeleportTo(zone.pos)
            task.wait(0.5)
            -- Nhận lại nhiệm vụ
            AutoQuest()
        end
    end
end
